cmake_minimum_required(VERSION 3.8)
project(mars_rover_systems)

if(CMAKE_COMPILER_IS_GNUCXX OR CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()

# find dependencies
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)

#create executable
add_executable(heartbeat_executable src/heartbeat.cpp)
add_executable(heartbeat_shutdown_executable src/heartbeat_shutdown.cpp)
add_executable(heartbeat_executable_2 src/heartbeat_2.cpp)
add_executable(temperature_executable src/temperature_monitor.cpp)

#add dependencies
ament_target_dependencies(heartbeat_executable rclcpp)
ament_target_dependencies(heartbeat_shutdown_executable rclcpp)
ament_target_dependencies(heartbeat_executable_2 rclcpp)
ament_target_dependencies(temperature_executable rclcpp)

#install executables
install(TARGETS 
	heartbeat_executable
	heartbeat_shutdown_executable
	heartbeat_executable_2
	temperature_executable
	DESTINATION lib/${PROJECT_NAME})

#install launch files
install(DIRECTORY launch 
	DESTINATION share/${PROJECT_NAME}/)

if(BUILD_TESTING)
  find_package(ament_lint_auto REQUIRED)
  # the following line skips the linter which checks for copyrights
  # comment the line when a copyright and license is added to all source files
  set(ament_cmake_copyright_FOUND TRUE)
  # the following line skips cpplint (only works in a git repo)
  # comment the line when this package is in a git repo and when
  # a copyright and license is added to all source files
  set(ament_cmake_cpplint_FOUND TRUE)
  ament_lint_auto_find_test_dependencies()
endif()

ament_package()
